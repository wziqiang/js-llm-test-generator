## 1. 基础算法类

这类函数逻辑清晰，适合展示大模型生成基本测试用例和覆盖所有分支的能力。

### 1.1 斐波那契数列 (Fibonacci)
演示点：递归边界条件（0和1）的处理。

```javascript
function fibonacci(n) {
  if (typeof n !== 'number' || n < 0) {
    throw new Error('Input must be a non-negative number');
  }
  if (n === 0) return 0;
  if (n === 1) return 1;
  return fibonacci(n - 1) + fibonacci(n - 2);
}

module.exports = { fibonacci };
```

**演示 Bug（失败场景）：**
将 `if (n === 0) return 0;` 改为 `if (n === 0) return 1;`，演示系统捕获错误 `Expected 1, Received 0` 的能力。

### 1.2 快速排序 (Quick Sort)
演示点：数组处理和基本算法逻辑。

```javascript
function quickSort(arr) {
  if (!Array.isArray(arr)) {
    throw new TypeError('Input must be an array');
  }
  if (arr.length <= 1) {
    return arr;
  }
  
  const pivot = arr[Math.floor(arr.length / 2)];
  const left = [];
  const right = [];
  const equal = [];
  
  for (let val of arr) {
    if (val < pivot) {
      left.push(val);
    } else if (val > pivot) {
      right.push(val);
    } else {
      equal.push(val);
    }
  }
  
  return [...quickSort(left), ...equal, ...quickSort(right)];
}

module.exports = { quickSort };
```

## 2. 字符串/数据处理类

这类函数边界条件较多（如空输入、非法格式），适合展示大模型思维链（CoT）推理边界情况的能力。

### 2.1 URL 参数解析 (Parse URL Parameters)
演示点：字符串处理、正则表达式、特殊字符解码。

```javascript
function parseUrlParams(url) {
  if (typeof url !== 'string') return {};
  
  const params = {};
  const queryString = url.split('?')[1];
  
  if (!queryString) return params;
  
  const pairs = queryString.split('&');
  
  for (let pair of pairs) {
    if (!pair) continue; // 处理 "&&" 的情况
    
    const [key, value] = pair.split('=');
    if (key) {
      // 处理没有 value 的情况，如 "?a"
      params[decodeURIComponent(key)] = value ? decodeURIComponent(value) : '';
    }
  }
  
  return params;
}

module.exports = { parseUrlParams };
```

### 2.2 对象深拷贝 (Deep Clone)
演示点：处理 null、数组、嵌套对象等不同数据类型。

```javascript
function deepClone(obj) {
  if (obj === null || typeof obj !== 'object') {
    return obj;
  }
  
  if (Array.isArray(obj)) {
    const copy = [];
    for (let i = 0; i < obj.length; i++) {
      copy[i] = deepClone(obj[i]);
    }
    return copy;
  }
  
  const copy = {};
  for (let key in obj) {
    if (obj.hasOwnProperty(key)) {
      copy[key] = deepClone(obj[key]);
    }
  }
  return copy;
}

module.exports = { deepClone };
```

## 3. 业务逻辑类

这类函数贴近实际业务，通常涉及数值计算精度或多重逻辑判断。

### 3.1 购物车金额计算 (Calculate Cart Total)
演示点：浮点数精度处理、折扣逻辑、异常数据容错。

```javascript
function calculateCartTotal(cartItems, discountCode = '') {
  if (!Array.isArray(cartItems) || cartItems.length === 0) {
    return 0;
  }

  let total = 0;

  for (const item of cartItems) {
    if (typeof item.price !== 'number' || typeof item.quantity !== 'number') {
      throw new Error('Invalid item data');
    }
    if (item.price < 0 || item.quantity < 0) {
      throw new Error('Price and quantity must be non-negative');
    }
    // 避免浮点数精度问题，先乘 100
    total += (item.price * 100) * item.quantity;
  }

  // 恢复正常的数值
  total = total / 100;

  // 处理折扣
  if (discountCode === 'VIP10') {
    total = total * 0.9;
  } else if (discountCode === 'MINUS50') {
    total = Math.max(0, total - 50); // 确保总价不为负
  }

  // 保留两位小数并转为数字
  return Number(total.toFixed(2));
}

module.exports = { calculateCartTotal };
```

### 3.2 用户权限判断 (Check User Permission)
演示点：多重逻辑分支覆盖。

```javascript
function checkUserPermission(user, resource, action) {
  if (!user || !user.role) return false;
  
  // 超级管理员拥有所有权限
  if (user.role === 'SUPER_ADMIN') {
    return true;
  }
  
  // 普通管理员只能读取和更新，不能删除
  if (user.role === 'ADMIN') {
    if (action === 'DELETE') return false;
    return true;
  }
  
  // 普通用户只能操作自己创建的资源
  if (user.role === 'USER') {
    if (!resource || !resource.ownerId) return false;
    if (resource.ownerId === user.id) {
      // 只能读取和更新自己的资源
      return ['READ', 'UPDATE'].includes(action);
    }
    return false;
  }
  
  return false;
}

module.exports = { checkUserPermission };
```
